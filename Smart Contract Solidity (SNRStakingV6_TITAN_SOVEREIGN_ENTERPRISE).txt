    function secureTreasuryTransfer() external onlyOwner nonReentrant {
        require(emergencyMode, "Emergency inactive");
        require(block.timestamp >= emergencyActivatedAt + 12 hours, "Security cooldown active");

        uint256 balance = snr.balanceOf(address(this));
        uint256 liabilities = totalLiabilities();
        
        uint256 currentBuffer = (liabilities * reserveBufferBP) / 10000;
        uint256 absoluteRequiredReserve = liabilities + currentBuffer;
        
        require(balance > absoluteRequiredReserve, "Insufficient assets above total liabilities and buffer");
        
        uint256 drainableAmount = balance - absoluteRequiredReserve;

        snr.safeTransfer(treasury, drainableAmount);
        emit EmergencyTreasurySecured(drainableAmount);
    }

    function joinProtocol(
        uint256 _amount,
        address _mentor,
        uint256 _days
    ) external nonReentrant whenNotPaused notLocked notFrozen hasSNRSovereignty {
        require(_amount >= 100 * 10**18, "Min 100 SNR");
        require(_days == 30 || _days == 60 || _days == 90, "Invalid duration");

        User storage u = users[msg.sender];
        require(u.participation + _amount <= maxStakeLimit, "Exceed max stake");

        if (leaderCapReached[msg.sender]) {
            leaderCapReached[msg.sender] = false;
            leaderRewardAccumulated[msg.sender] = 0;
            emit LeaderCapReset(msg.sender);
        }

        if (u.mentor == address(0)) {
            u.mentor = (_mentor != address(0) && (users[_mentor].isActive || _mentor == owner)) ? _mentor : rootWallet;
        }

        _updateUplineData(u.mentor, _amount);

        uint256 incomingYield = (_days == 30) ? 100 : (_days == 60 ? 120 : 150);

        if (u.isActive && u.participation > 0) {
            _internalHarvest(msg.sender, true);
            
            // CANON 1: Aturan Penggabungan Posisi Tunggal Terpilih via Weighted Average Formula
            uint256 currentWeight = u.participation * u.dailyYieldBP;
            uint256 incomingWeight = _amount * incomingYield;
            u.dailyYieldBP = (currentWeight + incomingWeight) / (u.participation + _amount);
        } else {
            u.dailyYieldBP = incomingYield;
        }

        snr.safeTransferFrom(msg.sender, address(this), _amount);

        u.isActive = true;
        u.participation += _amount;
        u.originalGrant += _amount;
        u.lastUpdate = block.timestamp;
        
        uint256 newExtDate = block.timestamp + (_days * 1 days);
        if (newExtDate > u.stakingEndDate) {
            u.stakingEndDate = newExtDate;
        }

        totalPrincipalLiability += _amount;
        _checkRank(msg.sender);
    }

    function _internalHarvest(address _user, bool force) internal {
        User storage u = users[_user];

        if (u.isBlacklisted || (!force && block.timestamp <= u.lastUpdate + 24 hours)) {
            return;
        }

        uint256 maxCap = (u.originalGrant * MAX_REWARD_CAP_PERCENT) / 100;
        uint256 available = _calculateGrowth(_user);

        if (available == 0 || u.totalRewardClaimed >= maxCap) {
            return;
        }

        uint256 remaining = maxCap - u.totalRewardClaimed;
        if (available > remaining) {
            available = remaining;
        }

        require(snr.balanceOf(address(this)) >= available, "Reserve tidak cukup");

        u.lastUpdate = block.timestamp;
        u.totalRewardClaimed += available;

        _distributeReferral(u.mentor, available);

        uint256 ecosystemFee = (available * 10) / 100;
        uint256 netReward = available - ecosystemFee;

        snr.safeTransfer(treasury, ecosystemFee);

        uint256 withdrawShare = netReward / 2;
        u.readyToWithdraw += withdrawShare;
        totalWithdrawLiability += withdrawShare;

        emit RewardHarvested(_user, 0, withdrawShare);
    }

    function _distributeReferral(address _mentor, uint256 _amount) internal {
        address current = _mentor;

        for (uint256 i = 0; i < 10; i++) {
            uint256 rewardShare = (_amount * referralLevels[i]) / 10000;

            if (current == address(0)) {
                leaderRewards[rootWallet] += rewardShare;
                totalReferralLiability += rewardShare;
                continue;
            }

            User storage leader = users[current];
            uint256 directCount = leader.directActiveCount;
            bool isQualified = false;

            if (i < 4) {
                isQualified = (directCount >= 1);
            } else if (i < 7) {
                isQualified = (directCount >= 5);
            } else {
                isQualified = (directCount >= 10);
            }

            uint256 maxLeaderCap = (leader.participation * MAX_LEADER_REWARD_PERCENT) / 100;
            bool capAvailable = leaderRewardAccumulated[current] < maxLeaderCap;

            if (isQualified && !leader.isBlacklisted && capAvailable && !leaderCapReached[current]) {
                uint256 remainingCap = maxLeaderCap - leaderRewardAccumulated[current];

                if (rewardShare > remainingCap) {
                    rewardShare = remainingCap;
                }

                leaderRewards[current] += rewardShare;
                leaderRewardAccumulated[current] += rewardShare;
                totalReferralLiability += rewardShare;

                if (leaderRewardAccumulated[current] >= maxLeaderCap) {
                    leaderCapReached[current] = true;
                    emit LeaderCapReached(current);
                }
            } else {
                leaderRewards[rootWallet] += rewardShare;
                totalReferralLiability += rewardShare;
            }

            current = users[current].mentor;
        }
    }

    function claimLeaderRewards() external nonReentrant notLocked notFrozen hasSNRSovereignty {
        uint256 reward = leaderRewards[msg.sender];
        require(reward > 0, "Reward kosong");
        require(snr.balanceOf(address(this)) >= reward, "Reserve tidak cukup");

        leaderRewards[msg.sender] = 0;
        leaderRewardsClaimed[msg.sender] += reward;

        if (totalReferralLiability >= reward) {
            totalReferralLiability -= reward;
        } else {
            totalReferralLiability = 0;
        }

        snr.safeTransfer(msg.sender, reward);
    }

    function withdrawReadyBalance(uint256 _amount) external nonReentrant notLocked notFrozen {
        User storage u = users[msg.sender];
        require(_amount > 0 && u.readyToWithdraw >= _amount, "Saldo tidak cukup");
        require(snr.balanceOf(address(this)) >= _amount, "Reserve tidak cukup");

        u.readyToWithdraw -= _amount;

        if (totalWithdrawLiability >= _amount) {
            totalWithdrawLiability -= _amount;
        } else {
            totalWithdrawLiability = 0;
        }

        uint256 wdFee = (_amount * 10) / 100;
        uint256 netToWallet = _amount - wdFee;

        snr.safeTransfer(treasury, wdFee);
        snr.safeTransfer(msg.sender, netToWallet);
    }

    function unstakePrincipal(uint256 _amount) external nonReentrant notLocked notFrozen {
        User storage u = users[msg.sender];
        require(_amount > 0, "Invalid amount");
        require(u.participation >= _amount, "Participation kurang");
        
        // TEMUAN 2 - BERBAHAYA: Kunci Jaminan Aset Mutlak. Modal pokok dilarang dicabut melewati batas pagu jaminan aktif!
        require(u.participation - _amount >= u.principalDeposit, "PROTEKSI: Saldo Staking tidak boleh kurang dari Jaminan Aset aktif!");
        
        require(snr.balanceOf(address(this)) >= _amount, "Reserve tidak cukup");

        uint256 penalty = 0;
        if (block.timestamp < u.stakingEndDate) {
            penalty = (_amount * 50) / 100;
        }

        uint256 netAmount = _amount - penalty;
        u.participation -= _amount;

        if (totalPrincipalLiability >= _amount) {
            totalPrincipalLiability -= _amount;
        } else {
            totalPrincipalLiability = 0;
        }

        if (u.originalGrant >= _amount) {
            u.originalGrant -= _amount;
        }

        if (u.participation == 0) {
            u.isActive = false;
            _deactivateDirect(msg.sender);
        }

        if (penalty > 0) {
            snr.safeTransfer(treasury, penalty);
        }

        snr.safeTransfer(msg.sender, netAmount);
        emit PrincipalUnstaked(msg.sender, netAmount, penalty);
    }

    function _deactivateDirect(address _user) internal {
        address mentor = users[_user].mentor;
        if (mentor != address(0) && users[mentor].directActiveCount > 0) {
            users[mentor].directActiveCount -= 1;
            hasContributedActive[mentor][_user] = false;
        }
    }

        AssetType _t,
        uint256 _v,
        string memory _model,

    ) external nonReentrant whenNotPaused notLocked notFrozen hasSNRSovereignty {
        User storage u = users[msg.sender];
        require(u.participation >= minStakeForAsset, "Need active staking");
        require(_v <= maxAssetValueLimit, "Exceeds absolute max asset parameter limit");

        if (_t == AssetType.MOTO || _t == AssetType.DRIVE) {
            require(u.participation >= minStakeForVehicle, "Need 3000 SNR");
        }

        require(u.activeProgram == AssetType.NONE, "Program active");

        if (_p == ProtocolPath.EXCLUSIVE_100) {
            u.principalDeposit = _v;
        } else {
            u.principalDeposit = (_v * 55) / 100;
        }

        // Kunci Integritas: Menghalangi pengajuan klaim program jika modal staking yang dimiliki tidak mencukupi nilai jaminan
        require(u.participation >= u.principalDeposit, "Jaminan melebihi saldo staking aktif");

        uint256 d;
        if (_t == AssetType.GOLD || _t == AssetType.MOTO) {
            d = 3 * 365 days;
        } else if (_t == AssetType.DRIVE || _t == AssetType.LIVING_STANDARD) {
            d = 5 * 365 days;
        } else {
            d = 10 * 365 days;
        }

        u.activeProgram = _t;
        u.supportValue = _v;
        u.assetEndDate = block.timestamp + d;

        userAssets[msg.sender] = AssetDetail(
            _model,
            "MOU_PENDING",
            block.timestamp,
            true,
            false,
            _p
        );

        emit AssetActivated(msg.sender, _t, _v, _p);
    }

    function claimSovereignPrincipal() external nonReentrant notLocked notFrozen {
        User storage u = users[msg.sender];
        require(u.activeProgram != AssetType.NONE, "No program");
        require(block.timestamp >= u.assetEndDate, "Locked");
        require(!u.isFundLockedByAdmin, "Fund locked");
        require(u.principalDeposit > 0, "Zero principal");

        uint256 amountToReturn = u.principalDeposit;
        u.principalDeposit = 0;
        u.activeProgram = AssetType.NONE;

        require(snr.balanceOf(address(this)) >= amountToReturn, "Reserve tidak cukup");
        snr.safeTransfer(msg.sender, amountToReturn);
    }

    function _updateUplineData(address _mentor, uint256 _amount) internal {
        address current = _mentor;
        for (uint256 i = 0; i < 10; i++) {
            if (current == address(0)) break;
            users[current].groupVolume += _amount;

            if (i == 0 && _amount >= minDirectStakeForActive) {
                if (!hasContributedActive[_mentor][msg.sender]) {
                    users[current].directActiveCount += 1;
                    hasContributedActive[_mentor][msg.sender] = true;
                }
            }
            _checkRank(current);
            current = users[current].mentor;
        }
    }

    function _checkRank(address _user) internal {
        User storage u = users[_user];
        if (!u.isActive || u.isBlacklisted) return;

        uint256 rank = 0;
        if (u.participation >= minDirectStakeForActive) {
            if (u.directActiveCount >= 100 && u.groupVolume >= 200000 * 10**18) {
                rank = 4;
            } else if (u.directActiveCount >= 50 && u.groupVolume >= 100000 * 10**18) {
                rank = 3;
            } else if (u.directActiveCount >= 15 && u.groupVolume >= 30000 * 10**18) {
                rank = 2;
            } else if (u.directActiveCount >= 5 && u.groupVolume >= 10000 * 10**18) {
                rank = 1;
            }

            if (rank > u.currentRank) {
                u.currentRank = rank;
                emit RankUpgraded(_user, rank);
            }
        }
    }

    function _calculateGrowth(address _user) internal view returns (uint256) {
        User storage u = users[_user];
        if (u.isBlacklisted) return 0;

        uint256 endPeriod = block.timestamp;
        if (endPeriod > u.stakingEndDate) {
            endPeriod = u.stakingEndDate;
        }
        if (endPeriod <= u.lastUpdate) {
            return 0;
        }
        return (u.participation * u.dailyYieldBP * (endPeriod - u.lastUpdate)) / (10000 * 1 days);
    }

function _currentReward(address _user)
    internal
    view
    returns (uint256)
{
    User storage u = users[_user];

    if (!u.isActive) {
        return 0;
    }

    uint256 reward = _calculateGrowth(_user);

    uint256 maxCap =
        (u.originalGrant * MAX_REWARD_CAP_PERCENT) / 100;

    if (u.totalRewardClaimed >= maxCap) {
        return 0;
    }

    uint256 remain =
        maxCap - u.totalRewardClaimed;

    if (reward > remain) {
        reward = remain;
    }

    return reward;
}

function _remainingLock(address _user)
    internal
    view
    returns (uint256)
{
    User storage u = users[_user];

    if (block.timestamp >= u.stakingEndDate) {
        return 0;
    }

    return u.stakingEndDate - block.timestamp;
}

function _totalStakingAsset(address _user)
    internal
    view
    returns (uint256)
{
    User storage u = users[_user];

    return
    u.participation +
    _currentReward(_user) +
    u.readyToWithdraw +
    leaderRewards[_user];
}

function _assetRemaining(address _user)
    internal
    view
    returns (uint256)
{
    User storage u = users[_user];

    if (u.activeProgram == AssetType.NONE) {
        return 0;
    }

    if (block.timestamp >= u.assetEndDate) {
        return 0;
    }

    return
        u.assetEndDate -
        block.timestamp;
}

function _rewardCapRemaining(address _user)
    internal
    view
    returns (uint256)
{
    User storage u = users[_user];

    uint256 maxCap =
        (u.originalGrant * MAX_REWARD_CAP_PERCENT) / 100;

    if (u.totalRewardClaimed >= maxCap) {
        return 0;
    }

    return
        maxCap -
        u.totalRewardClaimed;
}

    function totalLiabilities() public view returns (uint256) {
        return totalPrincipalLiability + totalReferralLiability + totalWithdrawLiability;
    }

    function reserveHealth() public view returns (uint256) {
        uint256 liabilities = totalLiabilities();
        if (liabilities == 0) return 10000; 
        return (snr.balanceOf(address(this)) * 10000) / liabilities;
    }

    function isSolvent() public view returns (bool) {
        return snr.balanceOf(address(this)) >= totalLiabilities();
    }

    function initiateTimelockUpdate(bytes32 _paramId, uint256 _newValue, address _targetAddress) external onlyOwner {
        // TEMUAN 5: Validasi antrean ketat. Mencegah overwrite paksa request timelock parameter yang sedang berjalan
        require(!adminTimelocks[_paramId].isPending, "Request timelock untuk parameter ini sedang berjalan");
        
        adminTimelocks[_paramId] = TimelockRequest({
            value: _newValue,
            targetAddress: _targetAddress,
            executionTime: block.timestamp + TIMELOCK_DELAY,
            isPending: true
        });
        emit TimelockInitiated(_paramId, _newValue, _targetAddress, block.timestamp + TIMELOCK_DELAY);
    }

    function executeMinDirectStakeForActive() external onlyOwner {
        bytes32 paramId = keccak256(abi.encodePacked("minDirectStakeForActive"));
        TimelockRequest storage r = adminTimelocks[paramId];
        require(r.isPending, "No request pending");
        require(block.timestamp >= r.executionTime, "Timelock delay active");
        
        minDirectStakeForActive = r.value * 10**18;
        r.isPending = false;
        emit ParameterUpdatedViaTimelock(paramId, r.value, address(0));
    }

    function rescueForeignToken(address _tokenAddress, address _to, uint256 _amount) external onlyOwner nonReentrant {
        require(_tokenAddress != address(0) && _to != address(0), "Invalid parameters");
        require(_tokenAddress != address(snr), "PROTEKSI: Token SNR dilarang dikuras lewat fungsi rescue!");

        IERC20(_tokenAddress).safeTransfer(_to, _amount);
        emit ForeignTokenRescued(_tokenAddress, _to, _amount);
    }

    // TWO-STEP SECURE OWNERSHIP TRANSITION
    // =====================================================

    function transferOwnership(address _newOwner) external onlyOwner {
        require(_newOwner != address(0), "Target wallet cannot be zero address");
        pendingOwner = _newOwner;
        emit OwnershipTransferStarted(owner, _newOwner);
    }

    function acceptOwnership() external {
        require(msg.sender == pendingOwner, "Caller is not authorized");
        address oldOwner = owner;
        owner = pendingOwner;
        pendingOwner = address(0);
        emit OwnershipTransferred(oldOwner, owner);
    }
